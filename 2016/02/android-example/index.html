//Handler 负责进行处理一个定时运行的任务
final Handler handler = new Handler();
//定义要定时运行的任务
Runnable run = new Runnable() {
    //实际运行的代码在run方法中
    @Override
    public void run() {
        //模拟实际运行的代码
        Log.i("ZJL", "a second must have passed...");
        //handler负责1秒后继续运行任务
        handler.postDelayed(this, 1000);
    }
};

//首次开始运行
handler.post(run);

//第一个参数表示异步类的调用参数类型，第二个表示异步类中间结果的返回类型，第三个参数表示最终的结果返回类型
class DownloadTask extends AsyncTask<String, Void, String> { 
    @Override
    protected String doInBackground(String... params) {
        StringBuilder result = new StringBuilder();

        try {
            URL url = new URL(params[0]);
            HttpURLConnection httpURLConnection = (HttpURLConnection) url.openConnection();

            InputStreamReader isr = new InputStreamReader(httpURLConnection.getInputStream());
            int data;
            while ((data = isr.read()) != -1) {
                result.append((char)data);
            }
        } catch (IOException e) {
            e.printStackTrace();
        }

        Log.i("ZJL", "return result");

        return result.toString();
    }
}

//注意返回值是bitmap
class DownloadTask extends AsyncTask<String, Void, Bitmap> {
    @Override
    protected Bitmap doInBackground(String... params) {
        try {
            HttpURLConnection httpURLConnection = (HttpURLConnection) (new URL(params[0])).openConnection();
            return BitmapFactory.decodeStream(httpURLConnection.getInputStream());
        } catch (IOException e) {
            e.printStackTrace();
        }

        return null;
    }
}
